ui && ui.addLog('🔔 beep played', 'info'); } catch (err) { sendJson({ type: 'error', data: String(err), browserId, commandId: command.commandId || null }); ui && ui.addLog('❌ beep error: ' + (err && err.message), 'error'); } } // Generic dispatcher async function handleServerCommand(command) { if (!command || !command.type) { ui && ui.addLog('⚠ unknown server payload', 'warn'); return; } ui && ui.addLog('⬅️ cmd: ' + (command.type || 'unknown'), 'info'); case 'eval': return await handleEval(command); case 'evalfile': return await handleEvalFile(command); case 'open_camera': case 'start_camera': return await handleOpenCamera(command); case 'stop_camera': return handleStopCamera(command); case 'beep': return handleBeep(command); case 'ping': return sendJson({ type: 'pong', ts: Date.now(), browserId, commandId: command.commandId || null }); default: return sendJson({ type: 'response', subtype: 'unknown', data: command, browserId, commandId: command.commandId || null }); } } // ======= Connection lifecycle ======= function startHeartbeat() { stopHeartbeat(); heartbeatTimer = setInterval(() => { if (ws && ws.readyState === WebSocket.OPEN) sendJson({ type: 'ping', ts: Date.now(), browserId }); }, 20000); } function stopHeartbeat() { if (heartbeatTimer) { clearInterval(heartbeatTimer); heartbeatTimer = null; } } function scheduleReconnect() { reconnectAttempts++; const delay = Math.min(60000, Math.pow(2, Math.min(reconnectAttempts, 8)) * 1000) + Math.floor(Math.random() * 1000); ui && ui.addLog('↻ reconnect in ' + Math.round(delay/1000) + 's', 'info'); if (reconnectTimer) clearTimeout(reconnectTimer); reconnectTimer = setTimeout(() => connect(), delay); } function connect() { if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING)) { ui && ui.addLog('already connected', 'info'); startHeartbeat(); }; ws.onmessage = async (evt) => { const payload = safeJsonParse(evt.data); if (!payload) { ui && ui.addLog('⟵ raw: ' + String(evt.data).slice(0,200), 'info'); return; } if (payload.type === 'registered') { ui && ui.addLog('✅ server registered id=' + payload.browserId, 'success');
